        ranked.sort(key=lambda item: (item[0], item[1].frequency, item[1].label), reverse=True)

        return [
            {
                "label": item.label,
                "kind": item.kind,
                "score": round(score, 4),
                "frequency": item.frequency,
                "document_frequency": item.document_frequency,
                "examples": item.examples,
            }
            for score, item in ranked[:limit]
        ]
